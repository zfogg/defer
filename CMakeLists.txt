cmake_minimum_required(VERSION 3.25)
project(defer VERSION 0.1.0 LANGUAGES CXX)
option(DEFER_STATIC_LIBCLANG "Link libclang and LLVM statically" OFF)
option(DEFER_FULLY_STATIC_EXECUTABLE "Link the executable without shared libraries (musl Linux)" OFF)
include(GNUInstallDirs)
include(CTest)
include(cmake/LLVMDevelopment.cmake)
defer_find_llvm_development()
find_package(Threads REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Defer.cmake")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Defer.cmake" DEFER_CMAKE_CONTENT)
configure_file(cmake/embed.h.in generated/defer_cmake.h @ONLY)
set(_core cpp/lib/api.cpp cpp/lib/driver.cpp cpp/lib/macro_expansion.cpp cpp/lib/manifest.cpp cpp/lib/plan.cpp cpp/lib/transform.cpp)
add_library(defer-objects OBJECT ${_core})
target_compile_features(defer-objects PUBLIC cxx_std_17)
target_include_directories(defer-objects PUBLIC cpp/include)
set_target_properties(defer-objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
defer_configure_llvm_host_tool(defer-objects)
target_link_libraries(defer-objects PUBLIC defer-llvm-support defer-libclang)
foreach(_kind STATIC SHARED)
  string(TOLOWER ${_kind} _suffix)
  add_library(defer-${_suffix} ${_kind} $<TARGET_OBJECTS:defer-objects>)
  set_target_properties(defer-${_suffix} PROPERTIES OUTPUT_NAME "defer-${_suffix}" WINDOWS_EXPORT_ALL_SYMBOLS ON)
  target_include_directories(defer-${_suffix} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cpp/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(defer-${_suffix} PUBLIC defer-llvm-support defer-libclang ${CMAKE_DL_LIBS})
endforeach()
add_executable(defer-tool cpp/src/main.cpp)
target_compile_features(defer-tool PRIVATE cxx_std_17)
target_include_directories(defer-tool PRIVATE cpp/include "${CMAKE_CURRENT_BINARY_DIR}/generated")
defer_configure_llvm_host_tool(defer-tool)
target_link_libraries(defer-tool PRIVATE defer-static Threads::Threads)
if(APPLE AND NOT DEFER_STATIC_LIBCLANG)
  set_property(TARGET defer-tool defer-shared APPEND PROPERTY BUILD_RPATH "$<TARGET_FILE_DIR:defer-libclang>")
  set_property(TARGET defer-tool defer-shared APPEND PROPERTY INSTALL_RPATH "$<TARGET_FILE_DIR:defer-libclang>")
endif()
if(DEFER_FULLY_STATIC_EXECUTABLE)
  if(NOT DEFER_STATIC_LIBCLANG OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "DEFER_FULLY_STATIC_EXECUTABLE requires static libclang on Linux")
  endif()
  target_link_options(defer-tool PRIVATE -static)
endif()
if(WIN32 AND NOT DEFER_STATIC_LIBCLANG)
  add_custom_command(TARGET defer-tool POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:defer-libclang>" "$<TARGET_FILE_DIR:defer-tool>")
endif()
if(BUILD_TESTING)
  foreach(_test api text clang transform scope_guard manifest)
    add_executable(defer-${_test}-test tests/${_test}_test.cpp)
    target_compile_features(defer-${_test}-test PRIVATE cxx_std_17)
    target_compile_definitions(defer-${_test}-test PRIVATE "DEFER_TEST_INCLUDE=\"${CMAKE_CURRENT_SOURCE_DIR}/cpp/include\"")
    defer_configure_llvm_host_tool(defer-${_test}-test)
    target_link_libraries(defer-${_test}-test PRIVATE defer-static)
    if(APPLE AND NOT DEFER_STATIC_LIBCLANG)
      set_property(TARGET defer-${_test}-test APPEND PROPERTY BUILD_RPATH "$<TARGET_FILE_DIR:defer-libclang>")
    endif()
    add_test(NAME defer-${_test} COMMAND defer-${_test}-test)
  endforeach()
  add_executable(defer-api-shared-test tests/api_test.cpp)
  target_compile_features(defer-api-shared-test PRIVATE cxx_std_17)
  target_compile_definitions(defer-api-shared-test PRIVATE "DEFER_TEST_INCLUDE=\"${CMAKE_CURRENT_SOURCE_DIR}/cpp/include\"")
  defer_configure_llvm_host_tool(defer-api-shared-test)
  target_link_libraries(defer-api-shared-test PRIVATE defer-shared)
  if(APPLE AND NOT DEFER_STATIC_LIBCLANG)
    set_property(TARGET defer-api-shared-test APPEND PROPERTY BUILD_RPATH "$<TARGET_FILE_DIR:defer-libclang>")
  endif()
  if(WIN32 AND NOT DEFER_STATIC_LIBCLANG)
    add_custom_command(TARGET defer-api-shared-test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:defer-libclang>" "$<TARGET_FILE_DIR:defer-api-shared-test>")
  endif()
  add_test(NAME defer-api-shared COMMAND defer-api-shared-test)
endif()
install(TARGETS defer-tool defer-static defer-shared RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY cpp/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES Defer.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/defer)
if(WIN32 AND NOT DEFER_STATIC_LIBCLANG)
  install(FILES "$<TARGET_FILE:defer-libclang>" DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
add_subdirectory(docs)
