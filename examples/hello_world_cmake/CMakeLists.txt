cmake_minimum_required(VERSION 3.25)
project(defer_example LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT DEFER_MODULE OR NOT DEFER_INCLUDE_DIR)
  message(FATAL_ERROR "Set DEFER_MODULE and DEFER_INCLUDE_DIR")
endif()
include("${DEFER_MODULE}")
add_executable(hello main.c)
target_include_directories(hello PRIVATE "${DEFER_INCLUDE_DIR}")
set(DEFER_NATIVE_BINARY "${DEFER_TOOL}")
defer_target(hello)
