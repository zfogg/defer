cmake_minimum_required(VERSION 3.25)
project(defer_example LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT RECAST_MODULE OR NOT RECAST_INCLUDE_DIR)
  message(FATAL_ERROR "Set RECAST_MODULE and RECAST_INCLUDE_DIR")
endif()
include("${RECAST_MODULE}")
add_executable(hello main.c)
target_include_directories(hello PRIVATE "${RECAST_INCLUDE_DIR}")
set(RECAST_NATIVE_BINARY "${RECAST_TOOL}")
recast_target(hello)
